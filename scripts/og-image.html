modest-auth

Sign-in for Node.
No dependencies.

Sign-in by emailed link, password, or passkey, with only Node's own crypto and SQLite.

server.ts
const auth = createAuth(db, {
  signInWith: ["link", "passkey"],
});
const handler = createHandler(auth, {
  origin: "https://example.org",
  linkPage: "/account/link",
  deliver: yourMailer,
});

createServer(async (req, res) => {
  if (await handleNode(handler, req, res)) return;
  const user = handler.userOf(toRequest(req));
  res.end(user ? `Hello, ${user.email}` : "Hello");
}).listen(3000);
